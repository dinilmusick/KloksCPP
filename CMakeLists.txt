cmake_minimum_required(VERSION 3.16)
project(KloksCppTimingMath LANGUAGES CXX)
include(CTest)
add_library(kloks_timing_math INTERFACE)
target_include_directories(kloks_timing_math INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(kloks_timing_math INTERFACE cxx_std_17)
if(BUILD_TESTING)
  add_executable(kloks_timing_math_test tests/timing_math_test.cpp)
  target_link_libraries(kloks_timing_math_test PRIVATE kloks_timing_math)
  if(MSVC)
    target_compile_options(kloks_timing_math_test PRIVATE /W4)
  else()
    target_compile_options(kloks_timing_math_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME kloks_timing_math_test COMMAND kloks_timing_math_test)
endif()
