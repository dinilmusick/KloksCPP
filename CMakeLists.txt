cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/version.txt" ULTRST_RELEASE_VERSION LIMIT_COUNT 1)
project(KloksCPP VERSION ${ULTRST_RELEASE_VERSION} LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(klokscpp INTERFACE)
add_library(KloksCPP::klokscpp ALIAS klokscpp)
target_include_directories(klokscpp INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(klokscpp INTERFACE cxx_std_17)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(TARGETS klokscpp EXPORT KloksCPPTargets)
install(EXPORT KloksCPPTargets NAMESPACE KloksCPP:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KloksCPP)
configure_package_config_file(cmake/KloksCPPConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/KloksCPPConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KloksCPP)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/KloksCPPConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/KloksCPPConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/KloksCPPConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/KloksCPP)
include(CTest)
if(BUILD_TESTING)
 foreach(name clock service protocol transport_clock meter transport_position timing_math)
  add_executable(${name}_test tests/${name}_test.cpp)
  target_link_libraries(${name}_test PRIVATE KloksCPP::klokscpp)
  if(MSVC)
    target_compile_options(${name}_test PRIVATE /W4 /WX)
  else()
    target_compile_options(${name}_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(name STREQUAL "transport_clock" OR name STREQUAL "meter")
    if(MSVC)
      target_compile_options(${name}_test PRIVATE /UNDEBUG)
    else()
      target_compile_options(${name}_test PRIVATE -UNDEBUG)
    endif()
  endif()
  add_test(NAME ${name} COMMAND ${name}_test)
 endforeach()
endif()

option(KLOKSCPP_TEST_NATIVE_FOUNDATIONS "Test pinned GutCPP and KrnlCPP adapters" OFF)
set(KLOKSCPP_GUTCPP_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/GutCPP/include" CACHE PATH "GutCPP public include path")
set(KLOKSCPP_KRNLCPP_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.deps/KrnlCPP/include" CACHE PATH "KrnlCPP public include path")
if(BUILD_TESTING AND KLOKSCPP_TEST_NATIVE_FOUNDATIONS)
 if(NOT EXISTS "${KLOKSCPP_GUTCPP_INCLUDE_DIR}/gut/result.hpp" OR
    NOT EXISTS "${KLOKSCPP_KRNLCPP_INCLUDE_DIR}/krnl/kernel.hpp")
  message(FATAL_ERROR "Real GutCPP and KrnlCPP headers are required")
 endif()
 add_executable(native_adapter_test tests/native_adapter_test.cpp)
 target_link_libraries(native_adapter_test PRIVATE KloksCPP::klokscpp)
 target_include_directories(native_adapter_test PRIVATE "${KLOKSCPP_GUTCPP_INCLUDE_DIR}" "${KLOKSCPP_KRNLCPP_INCLUDE_DIR}")
 if(MSVC)
  target_compile_options(native_adapter_test PRIVATE /W4 /WX /permissive- /utf-8)
 else()
  target_compile_options(native_adapter_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
 add_test(NAME native_adapter COMMAND native_adapter_test)
endif()
