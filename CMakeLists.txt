cmake_minimum_required(VERSION 3.16)
project(KloksCPP LANGUAGES CXX)
add_library(klokscpp INTERFACE)
target_include_directories(klokscpp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(klokscpp INTERFACE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
 add_executable(transport_clock_test tests/transport_clock_test.cpp)
 target_link_libraries(transport_clock_test PRIVATE klokscpp)
 if(MSVC)
  target_compile_options(transport_clock_test PRIVATE /W4 /WX)
 else()
  target_compile_options(transport_clock_test PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
 add_test(NAME transport_clock COMMAND transport_clock_test)
endif()
