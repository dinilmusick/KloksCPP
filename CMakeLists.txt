cmake_minimum_required(VERSION 3.16)
project(KloksCPP LANGUAGES CXX)
add_library(klokscpp INTERFACE)
target_include_directories(klokscpp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(klokscpp INTERFACE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
 add_executable(transport_clock_test tests/transport_clock_test.cpp)
 target_link_libraries(transport_clock_test PRIVATE klokscpp)
 add_executable(meter_test tests/meter_test.cpp)
 add_executable(transport_position_test tests/transport_position_test.cpp)
 target_link_libraries(meter_test PRIVATE klokscpp)
 foreach(test_target transport_clock_test meter_test transport_position_test)
  if(MSVC)
   target_compile_options(${test_target} PRIVATE /W4 /WX)
  else()
   target_compile_options(${test_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
 endforeach()
 add_test(NAME transport_clock COMMAND transport_clock_test)
 add_test(NAME meter COMMAND meter_test)
 add_test(NAME transport_position COMMAND transport_position_test)
endif()
